    recovery_conscious BOOLEAN,                                -- 是否恢复躁动（清醒）
    skin_condition VARCHAR(50),                                -- 皮肤黏膜状况
    steward_score INTEGER,                                     -- Steward苏醒评分
    awakening_level VARCHAR(50),                               -- 清醒程度
    airway_patency VARCHAR(50),                                -- 呼吸道通畅程度
    limb_activity VARCHAR(50),                                 -- 肢体活动度

    pupil_light_reflex VARCHAR(50),                            -- 瞳孔光反射

    muscle_strength INTEGER,                                   -- 肌力
    top_ratio INTEGER,                                         -- TOF（神经肌肉监测指标）
